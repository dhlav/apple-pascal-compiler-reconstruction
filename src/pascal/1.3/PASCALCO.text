{ Apple Pascal 1.3 SYSTEM.COMPILER, segment 1 (PASCALCO).

  This is the whole segment's procedure list, in Apple's own order. The
  order is not cosmetic: UCSD numbers a procedure when it parses the
  header, so declaration order IS the numbering, and the numbering is what
  a `CGP n` in someone else's body has to agree with. Get the order wrong
  and every call site is wrong even if every body is right.

  The order comes from UCSD II.0's forward-declaration block
  (compglbls.text:362-402), which lines up with Apple's numbers 4..26 with
  four insertions and one deletion -- and every II.0 signature in it
  reproduces Apple's PARAM SIZE exactly. See finding 61.

  Procedures still to be reconstructed carry (*STUB*) on their header and
  have an empty body. They are here to hold their number, nothing more.
  The marker disappears from a procedure when its body is written and
  verified. }

PROCEDURE GETBOUNDS(*FSP: STP; VAR FMIN,FMAX: INTEGER*);
BEGIN
  WITH FSP^ DO
    IF FORM = SUBRANGE THEN
      BEGIN FMIN := MIN.IVAL; FMAX := MAX.IVAL END
    ELSE
      BEGIN FMIN := 0;
        IF FSP = CHARPTR THEN FMAX := 255
        ELSE
          IF FSP^.FCONST <> NIL THEN
            FMAX := FSP^.FCONST^.VALUES.IVAL
          ELSE FMAX := 0
      END
END;

PROCEDURE BUMPSEG(*VAR FVALUE: INTEGER; FMAX,FERRNUM: INTEGER*);
BEGIN
  IF FVALUE < FMAX THEN FVALUE := FVALUE+1
  ELSE ERROR(FERRNUM)
END;

PROCEDURE NEWSEG(*FNEWSEG: BOOLEAN*);
BEGIN
  IF FNEWSEG THEN
    BEGIN BUMPSEG(NEXTSEG,63,354); BUMPSEG(SEGSLOT,15,354) END;
  SEGMAP[SEG] := SEGSLOT
END;

PROCEDURE SKIP(*FSYS: SETOFSYS*);
BEGIN WHILE NOT(SY IN FSYS) DO INSYMBOL
END;

FUNCTION PAOFCHAR(*FSP: STP): BOOLEAN*);
BEGIN PAOFCHAR := FALSE;
  IF FSP <> NIL THEN
    IF FSP^.FORM = ARRAYS THEN
      PAOFCHAR := FSP^.AISPACKD AND (FSP^.AELTYPE = CHARPTR)
END;

FUNCTION STRGTYPE(*FSP: STP): BOOLEAN*);
BEGIN STRGTYPE := FALSE;
  IF PAOFCHAR(FSP) THEN STRGTYPE := FSP^.AISSTRNG
END;

FUNCTION DECSIZE(*I: INTEGER): INTEGER*);
BEGIN DECSIZE := (I + 3) DIV 4 + 1
END;

PROCEDURE GENBYTE(*FBYTE: INTEGER*);
BEGIN
  CODEP^[IC] := CHR(FBYTE); IC := IC+1
END;

PROCEDURE GENWORD(*FWORD: INTEGER*);
  VAR TEMP: CHAR;
BEGIN
  IF ODD(IC) THEN GENBYTE(0);
  MOVELEFT(FWORD,CODEP^[IC],2);
  IF FLIPBYTES THEN
    BEGIN
      TEMP := CODEP^[IC];
      CODEP^[IC] := CODEP^[IC+1];
      CODEP^[IC+1] := TEMP
    END;
  IC := IC + 2
END;
