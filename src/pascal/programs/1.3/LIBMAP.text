{ LIBMAP -- the Apple Pascal Library Map utility, APPLE2:LIBMAP.CODE.

  Reconstructed from the 1.3 binary. Eleven p-code procedures and one
  native routine -- and the native routine needs no new source at all: it
  is IDSEARCH, the compiler's own reserved-word lookup, linked in whole
  (finding 109). `PROCEDURE IDSEARCH(VAR IDREC, ID); EXTERNAL;` declared
  first, exactly as PASCALCO.text declares it, and SYSTEM.LINKER puts the
  same assembled procedure from src/native/SEARCH.TEXT in its place.

  Procedure numbers are fixed by declaration order and the binary fixes
  the order: IDSEARCH is 2, then 3 through 7 and 12 at the outer level,
  with 8 nested inside 7's own declaration part (lex 2), and 9, 10, 11
  nested inside 8 (lex 3) -- the lex levels in the p-code listing are the
  nesting, not a guess.

  The one segment dictionary LIBMAP reads is Apple's own on-disk format
  (block 0 of a library or codefile): addr/len per slot, then names, then
  segkind, then a table read but never named elsewhere in this file, then
  SEGINFO, seventy-two words this program never references at all, and a
  length-prefixed comment -- 256 words exactly, one disk block. Every
  field offset below is the LAO/LDO operand the binary uses, not a guess.

  Globals. SEGDICT is the whole dictionary block, INFILE the library being
  mapped, OUTFILE the map's destination. Four booleans and two integer
  tables (version strings, machine-type strings) follow, then two more
  integers the outer block's loop uses. Every name is ours; the offsets
  are Apple's. }

{ $I- : ordinary WRITE/WRITELN/RESET on OUTFILE and INFILE compile to the
  same CXP 0,n the binary calls directly (verified: WRITE(OUTFILE,S) and
  FWRITESTRING(@OUTFILE,S,0) are the same instructions), so this does not
  need (*$U-*) or a single OS FORWARD -- only IDSEARCH is EXTERNAL. Without
  $I- every one of those calls carries an extra IOCHECK CSP the binary
  does not have. }
(*$I-*)
PROGRAM LIBMAP;

  TYPE
    DIRENTRY = RECORD
                 ADDR, LENG: INTEGER
               END;
    SEGINFOREC = PACKED RECORD CASE INTEGER OF
                   0: (ASWORD: INTEGER);
                   1: (SEGNUM: 0..255; MTYPE: 0..15;
                       UNUSED: 0..1; VERSION: 0..7)
                 END;
    SEGDICTREC = RECORD
                   ADDRLEN:  ARRAY [0..15] OF DIRENTRY;      {   3 }
                   SEGNAME:  ARRAY [0..15] OF
                               PACKED ARRAY [1..8] OF CHAR;  {  35 }
                   SEGKIND:  ARRAY [0..15] OF INTEGER;       {  99 }
                   TEXTADR:  ARRAY [0..15] OF INTEGER;       { 115 }
                   SEGINFO:  ARRAY [0..15] OF SEGINFOREC;    { 131 }
                   SPARE:    ARRAY [1..72] OF INTEGER;       { 147, unread }
                   COMMENT:  STRING[79]                     { 219 }
                 END;

  VAR
    SEGDICT:  SEGDICTREC;              {   3, one disk block }
    INFILE:   FILE;                    { 259, the library being mapped }
    OUTFILE:  TEXT;                    { 299, the map's destination }
    LISTMAP,                           { 600 }
    LISTREFS,                          { 601 }
    USINGLIB,                          { 602 }
    DONEFLAG: BOOLEAN;            { 603 }
    VERSTRS:  ARRAY [0..7] OF STRING[7];    { 604, 'unknown'/'A2/1.n' }
    TYPESTRS: ARRAY [0..9] OF STRING[23];   { 636, machine-type names }
    TEMP2, TEMP1: INTEGER;             { 756, 757 }

  PROCEDURE IDSEARCH(VAR IDREC, ID);
    EXTERNAL;

  { Byte-swap a word in place -- read from a library the current machine
    did not write, whose words come the other way round (the type table
    at G636 spells out both: "P-Code (most sig. 1st)" and "(least sig.
    1st)"). Verified instruction for instruction against the binary,
    LDP/STP field offsets included. }
  PROCEDURE SWAPBYTES(VAR W: INTEGER);
    VAR
      OLDVAL, NEWVAL: PACKED RECORD CASE BOOLEAN OF
                        TRUE:  (ASWORD: INTEGER);
                        FALSE: (LO, HI: 0..255)
                      END;
  BEGIN
    OLDVAL.ASWORD := W;
    NEWVAL.HI := OLDVAL.LO;
    NEWVAL.LO := OLDVAL.HI;
    W := NEWVAL.ASWORD
  END;

  { TRUE if any of the sixteen words at P has a nonzero byte on the side
    a word's high byte would be if this machine mis-swapped it -- the
    test for "this library's dictionary was written on a machine with
    the other byte order". The endianness probe (set a word to 1, ask
    which byte holds it) is generic UCSD, not an Apple-specific trick;
    on the 6502 it always answers "byte 0". Not yet checked against the
    binary's own CHK placement on the constant index -- (*STUB*) in
    spirit, real in shape. }
  FUNCTION NEEDSSWAP(VAR P: ARRAY [0..15] OF INTEGER): BOOLEAN;
    TYPE
      WORDBYTES = PACKED RECORD CASE BOOLEAN OF
                    TRUE:  (ASWORD: INTEGER);
                    FALSE: (ASBYTES: PACKED ARRAY [0..1] OF 0..255)
                  END;
    VAR
      HIBYTE: 0..1;
      I, LAST: 0..15;
      W: WORDBYTES;
  BEGIN
    NEEDSSWAP := FALSE;
    W.ASWORD := 1;
    HIBYTE := ORD(W.ASBYTES[0] = 1);
    I := 0;
    LAST := 15;
    WHILE I <= LAST DO
    BEGIN
      W.ASWORD := P[I];
      IF W.ASBYTES[HIBYTE] <> 0 THEN
      BEGIN
        NEEDSSWAP := TRUE
      END;
      I := I + 1
    END
  END;

  { Byte-swap every word in one dictionary block that came from a
    different-endian machine: both words of each ADDRLEN entry, then
    SEGKIND, TEXTADR and SEGINFO entry by entry. SEGKIND and SEGINFO go
    through a temporary and are written back; ADDRLEN and TEXTADR are
    swapped in place directly. Reconstructed from the shape of the
    binary's own loop; its own two extra words of locals (an address
    computed into its own temporary before SWAPBYTES dereferences it,
    same as NEEDSSWAP) are not yet accounted for here. }
  PROCEDURE SWAPALL(VAR D: SEGDICTREC);
    VAR
      I, LAST: 0..15;
      TEMP: INTEGER;
      INFOTEMP: SEGINFOREC;
  BEGIN
    I := 0;
    LAST := 15;
    WHILE I <= LAST DO
    BEGIN
      SWAPBYTES(D.ADDRLEN[I].ADDR);
      SWAPBYTES(D.ADDRLEN[I].LENG);
      TEMP := D.SEGKIND[I];
      SWAPBYTES(TEMP);
      D.SEGKIND[I] := TEMP;
      SWAPBYTES(D.TEXTADR[I]);
      INFOTEMP := D.SEGINFO[I];
      SWAPBYTES(INFOTEMP.ASWORD);
      D.SEGINFO[I] := INFOTEMP;
      I := I + 1
    END
  END;

  { TRUE iff all eight characters of a segment name are valid identifier
    characters (letter, digit, underscore or the blank padding) --
    checked before trusting a segment name at all. Verified instruction
    for instruction, including the early exit: the source leaves the
    result FALSE from the top and jumps straight past the loop rather
    than reassigning it. }
  FUNCTION VALIDNAME(VAR NAME: PACKED ARRAY [0..7] OF CHAR): BOOLEAN;
    LABEL 1;
    VAR
      LAST, I: 0..7;
  BEGIN
    VALIDNAME := FALSE;
    I := 0;
    LAST := 7;
    WHILE I <= LAST DO
    BEGIN
      IF NOT (NAME[I] IN [' ', '0'..'9', 'A'..'Z', '_']) THEN
      BEGIN
        GOTO 1
      END;
      I := I + 1
    END;
    VALIDNAME := TRUE;
  1:
  END;

  { Walk all sixteen dictionary slots and print the ones actually in use
    (nonzero code address or length). Each gets a header -- SEGINFO's own
    segment number if it has one, the slot position if not (an older,
    pre-II.1 file has no SEGINFO at all) -- then the version/machine-type
    line or "Pre-II.1 segment", then SHOWONE's own detail, then a rule.
    Reconstructed from the shape of the binary; not yet checked
    instruction for instruction. }
  PROCEDURE SHOWSEGS;
    VAR
      SEGNUM, LASTSEG: 0..16;
      ENTRY: ^DIRENTRY;

    PROCEDURE SHOWONE(SEGNUM: INTEGER);
      (*STUB*)

      FUNCTION SHOWINFO(BLK: INTEGER): BOOLEAN;
        (*STUB*)
      BEGIN
      END;

      PROCEDURE GETWORD(VAR W: INTEGER);
        (*STUB*)
      BEGIN
      END;

      PROCEDURE SHOWREF(VAR NAME: STRING);
        (*STUB*)
      BEGIN
      END;

    BEGIN
    END;

  BEGIN
    SEGNUM := 0;
    LASTSEG := 15;
    WHILE SEGNUM <= LASTSEG DO
    BEGIN
      ENTRY := @SEGDICT.ADDRLEN[SEGNUM];
      IF (ENTRY^.LENG > 0) OR (ENTRY^.ADDR > 0) THEN
      BEGIN
        IF SEGDICT.SEGINFO[SEGNUM].SEGNUM <> 0 THEN
        BEGIN
          WRITE(OUTFILE, 'Segment #');
          WRITE(OUTFILE, SEGDICT.SEGINFO[SEGNUM].SEGNUM:2);
          WRITE(OUTFILE, ':  ');
          WRITELN(OUTFILE)
        END
        ELSE
        BEGIN
          WRITE(OUTFILE, 'Segment #');
          WRITE(OUTFILE, SEGNUM);
          WRITE(OUTFILE, ':  ');
          WRITELN(OUTFILE)
        END;
        IF SEGDICT.SEGINFO[SEGNUM].VERSION <> 0 THEN
        BEGIN
          WRITE(OUTFILE, 'System version = ');
          WRITE(OUTFILE, VERSTRS[SEGDICT.SEGINFO[SEGNUM].VERSION]);
          WRITE(OUTFILE, ', code type is ');
          WRITE(OUTFILE, TYPESTRS[SEGDICT.SEGINFO[SEGNUM].MTYPE]);
          WRITELN(OUTFILE)
        END
        ELSE
        BEGIN
          WRITE(OUTFILE, 'Pre-II.1 segment');
          WRITELN(OUTFILE)
        END;
        SHOWONE(SEGNUM);
        WRITELN(OUTFILE);
        WRITE(OUTFILE, '----------------------------------------------------------------------':75);
        WRITELN(OUTFILE);
        WRITELN(OUTFILE)
      END;
      SEGNUM := SEGNUM + 1
    END
  END;

  PROCEDURE MAPLIBRARY;
    (*STUB*)
  BEGIN
  END;

BEGIN
END.
