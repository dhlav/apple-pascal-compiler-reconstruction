PROGRAM STRTEST;
VAR F: FILE; T: STRING[40]; U: STRING; N: INTEGER; C: CHAR;

PROCEDURE P1; BEGIN DELETE(T,1,40) END;
PROCEDURE P2; BEGIN INSERT(U,T,1) END;
PROCEDURE P3; BEGIN N := POS('.',T) END;
PROCEDURE P4; BEGIN N := LENGTH(T) END;
PROCEDURE P5; BEGIN STR(N,T) END;
PROCEDURE P6; BEGIN U := T END;
PROCEDURE P7; BEGIN RESET(F,U) END;
PROCEDURE P8; VAR L: STRING[80]; BEGIN RESET(F,L) END;
PROCEDURE P9; BEGIN C := T[1] END;
PROCEDURE PA; BEGIN T := '' END;

BEGIN
  P1; P2; P3; P4; P5; P6; P7; P8; P9; PA
END.
