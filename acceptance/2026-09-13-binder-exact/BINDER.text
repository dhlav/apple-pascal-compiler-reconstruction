{ BINDER -- "APPLE GOTOXY BINDER", APPLE3:BINDER.CODE, reconstructed.

  Six procedures in one segment, APPLEBIN. It reads a codefile whose
  procedure 2 is a native GOTOXY, copies that procedure into segment 15 of
  SYSTEM.PASCAL, shifts segment 0's link information to make room, and
  writes the result as NEW.PASCAL.

  No source survives. UCSD's I.5 BINDER is a different program (it links
  code), so nothing here is borrowed: the names are this reconstruction's,
  for roles the bodies show. All six procedures are instruction- and
  frame-identical to Apple's (finding 274). The file itself is a 1.1
  binary Apple carried into 1.3 unchanged, with version 2 in its SEGINFO
  word, which a 1.3 compile cannot write (finding 99c).

  No CHK and no CSP 0 anywhere: range and I/O checking are both off. }

(*$R-*)
(*$I-*)
PROGRAM APPLEBIN;

  TYPE
    DIRENTRY = RECORD LENG, ADDR: INTEGER END;   { ADDR 0, LENG 1 }
    { A dictionary is read as one block; nothing past ADDRLEN is touched,
      and the 512-word span is what the globals' offsets require. }
    SEGDICT = RECORD
                ADDRLEN: ARRAY [0..15] OF DIRENTRY;
                REST:    ARRAY [1..480] OF INTEGER
              END;
    BYTEBUF = PACKED ARRAY [0..511] OF 0..255;
    { SYSLEN is read whole and written back a byte at a time. }
    TWOBYTES = RECORD CASE BOOLEAN OF
                 TRUE:  (INT:  INTEGER);
                 FALSE: (BYTE: PACKED ARRAY [0..1] OF 0..255)
               END;

  VAR
    RESERVE:    ^BYTEBUF;    { 3: allocated only to count free pages }
    PATCH:      ^BYTEBUF;    { 4: the working buffer, a chain of pages }
    PATCHSTART: ^BYTEBUF;    { 5: its first page: all of SYSTEM.PASCAL }
    GTXDICT:    SEGDICT;     { 6: the GOTOXY file's dictionary }
    DICT:       SEGDICT;     { 518: SYSTEM.PASCAL's, becomes NEW.PASCAL's }
    SYSLEN:     TWOBYTES;    { 1030 }
    PATCHEND:   INTEGER;     { 1031: segment 15's new length }
    ENDBLOCKS:  INTEGER;     { 1032 }
    SYSBLOCKS:  INTEGER;     { 1033: segment 15's old length in blocks }
    PATCHSIZE:  INTEGER;     { 1034: GOTOXY's size in bytes }
    NEXTBLK:    INTEGER;     { 1035: NEW.PASCAL's next free block }
    SEGNUM:     INTEGER;     { 1036 }
    PROCCOUNT:  INTEGER;     { 1037: segment 15's procedure count }
    SEG0SHIFT:  INTEGER;     { 1038 }
    BLOCKS:     INTEGER;     { 1039: pages the heap can still hold }
    OSFILE, NEWFILE: FILE;   { NEWFILE 1040, OSFILE 1080 }

  { Halt with MSG if FAILED. }
  PROCEDURE CHECKERR(FAILED: BOOLEAN; MSG: STRING);
    VAR
      KEY: CHAR;
  BEGIN
    IF FAILED THEN
      BEGIN
        WRITELN;
        WRITELN;
        WRITE('ERROR: ',CHR(7),MSG);
        WRITELN;
        WRITELN;
        WRITE('Press space to continue');
        KEY := CHR(63);
        WHILE KEY <> ' ' DO READ(KEYBOARD,KEY);
        EXIT(PROGRAM)
      END
  END;

  { The word at byte OFS of PATCH^, low byte first. }
  FUNCTION WORDAT(OFS: INTEGER): INTEGER;
  BEGIN
    WORDAT := PATCH^[OFS] + 256 * PATCH^[OFS + 1]
  END;

  { Ask for the codefile, find procedure 2 in segment 0 or 1, and leave
    just that procedure and its jump-table word at the start of PATCH^. }
  PROCEDURE READGOTOXY;
    VAR
      PROCOFS, PATCHBASE, NUMBLOCKS, SEGSTART: INTEGER;
      GTXFILE: FILE;
      NAME:    STRING[80];
  BEGIN
    REPEAT
      IF IORESULT <> 0 THEN WRITE(CHR(7),CHR(7),CHR(7));
      WRITE('FILE WHICH CONTAINS GOTOXY? ');
      READLN(NAME);
      IF LENGTH(NAME) = 0 THEN EXIT(PROGRAM);
      RESET(GTXFILE,NAME);
      IF IORESULT <> 0 THEN RESET(GTXFILE,CONCAT(NAME,'.CODE'))
    UNTIL IORESULT = 0;
    CHECKERR(BLOCKREAD(GTXFILE,GTXDICT,1,0) <> 1,
             'READING SEG DICT OF GOTOXY FILE');
    IF GTXDICT.ADDRLEN[0].LENG <> 0 THEN
      SEGSTART := 0
    ELSE
      IF GTXDICT.ADDRLEN[1].LENG <> 0 THEN
        SEGSTART := 1
      ELSE
        CHECKERR(TRUE,'GOTOXY MUST BE IN SEG #0 OR #1');
    WITH GTXDICT.ADDRLEN[SEGSTART] DO
      BEGIN
        NUMBLOCKS := (LENG + 511) DIV 512;
        CHECKERR(BLOCKREAD(GTXFILE,PATCH^,NUMBLOCKS,ADDR) <> NUMBLOCKS,
                 'READING GOTOXY');
        PROCOFS := LENG - 6
      END;
    CHECKERR(PATCH^[PROCOFS - 1] <= 1,'GOTOXY MUST BE PROCEDURE #2');
    CHECKERR(WORDAT(PROCOFS) <= 0,'NO PROCEDURE #2 IN CODE FILE');
    PROCOFS := PROCOFS - WORDAT(PROCOFS);
    PATCHBASE := (PROCOFS - 2) - WORDAT(PROCOFS - 2);
    PATCHSIZE := PROCOFS - PATCHBASE;
    MOVELEFT(PATCH^[PATCHBASE],PATCH^[0],PATCHSIZE + 2);
    CLOSE(GTXFILE)
  END;

  { Write PATCH^'s first LEN bytes to NEWFILE as segment SEG. }
  PROCEDURE WRITESEG(LEN, SEG: INTEGER);
    VAR
      NUMBLOCKS: INTEGER;
  BEGIN
    IF LEN <> 0 THEN
      BEGIN
        WRITELN('  COPYING SEGMENT ',SEG);
        NUMBLOCKS := (LEN + 511) DIV 512;
        CHECKERR(BLOCKWRITE(NEWFILE,PATCH^,NUMBLOCKS,NEXTBLK) <> NUMBLOCKS,
                 'WRITING OUT SEGMENT');
        WITH DICT.ADDRLEN[SEG] DO
          BEGIN
            ADDR := NEXTBLK;
            LENG := LEN
          END;
        NEXTBLK := NEXTBLK + NUMBLOCKS
      END
  END;

  { Copy segment SEG out of the resident SYSTEM.PASCAL into PATCH^. }
  PROCEDURE LOADSEG(SEG: INTEGER);
    VAR
      NUMBLOCKS: INTEGER;
  BEGIN
    WITH DICT.ADDRLEN[SEG] DO
      BEGIN
        NUMBLOCKS := (LENG + 511) DIV 512;
        MOVERIGHT(PATCHSTART^[512 * (ADDR - 1)],PATCH^[0],LENG)
      END
  END;

BEGIN
  WRITELN;
  WRITELN;
  WRITELN;
  WRITELN;
  WRITELN;
  WRITE('':12,'APPLE GOTOXY BINDER');
  WRITELN;
  WRITELN;
  WRITELN;
  WRITELN;
  WRITELN;
  WRITELN;
  RESET(OSFILE,'SYSTEM.PASCAL');
  CHECKERR(IORESULT <> 0,'NO FILE SYSTEM.PASCAL');
  CHECKERR(BLOCKREAD(OSFILE,DICT,1,0) <> 1,'READING SEG DICT OF O.S.');
  NEW(PATCH);
  PATCHSTART := PATCH;
  WHILE BLOCKREAD(OSFILE,PATCH^,1) = 1 DO NEW(PATCH);
  BLOCKS := 1;
  WHILE MEMAVAIL > 2500 DO
    BEGIN
      NEW(RESERVE);
      BLOCKS := BLOCKS + 1
    END;
  READGOTOXY;
  NEXTBLK := 1;
  REWRITE(NEWFILE,'NEW.PASCAL');
  WITH DICT.ADDRLEN[15] DO
    BEGIN
      SYSLEN.INT := LENG;
      SYSBLOCKS := (LENG + 511) DIV 512;
      PATCHEND := LENG + PATCHSIZE + 2;
      ENDBLOCKS := (PATCHEND + 511) DIV 512;
      CHECKERR(ENDBLOCKS > BLOCKS,'NOT ENOUGH ROOM TO READ SEGMENT');
      CHECKERR(BLOCKREAD(OSFILE,PATCH^[PATCHSIZE + 2],SYSBLOCKS,ADDR)
               <> SYSBLOCKS,'READING O.S. SEG #15')
    END;
  PROCCOUNT := PATCH^[PATCHEND - 2];
  PATCH^[PATCHSIZE] := 29;
  PATCH^[PATCHSIZE + 1] := 0;
  WRITESEG(PATCHEND,15);
  CLOSE(OSFILE);
  WITH DICT.ADDRLEN[0] DO
    BEGIN
      LOADSEG(0);
      SEG0SHIFT := (LENG - 2) - 2 * PROCCOUNT;
      SYSLEN.INT := SYSLEN.INT + WORDAT(SEG0SHIFT);
      SYSLEN.INT := SYSLEN.INT + 2 * (PROCCOUNT - 29);
      PATCH^[(LENG - 2) - 2 * 29] := SYSLEN.BYTE[0];
      PATCH^[(LENG - 2) - 2 * 29 + 1] := SYSLEN.BYTE[1];
      WRITESEG(LENG,0)
    END;
  FOR SEGNUM := 1 TO 14 DO
    WITH DICT.ADDRLEN[SEGNUM] DO
      BEGIN
        LOADSEG(SEGNUM);
        WRITESEG(LENG,SEGNUM)
      END;
  CHECKERR(BLOCKWRITE(NEWFILE,DICT,1,0) <> 1,'WRITING SEG DICT OF NEW O.S.');
  CLOSE(NEWFILE,LOCK)
END.
